<?php
// ============================================================
//  EKSTRAK ojs-3.1.2-1.tar.gz MENJADI FOLDER ojs-3.1.2-1
//  Tanpa exec / shell / system
// ============================================================
set_time_limit(0);
ini_set('memory_limit', '512M');

$message = '';
$targetDir = __DIR__; // direktori skrip ini berada

// Path default file arsip (ada di folder yang sama dengan skrip)
$defaultArchivePath = $targetDir . '/ojs-3.1.2-1.tar.gz';
$archivePath = isset($_POST['file_path']) && !empty($_POST['file_path']) 
               ? $_POST['file_path'] 
               : $defaultArchivePath;

// Folder tujuan ekstrak (otomatis sesuai nama file tanpa .tar.gz)
$extractTo = $targetDir . '/ojs-3.1.2-1';

if (isset($_POST['extract'])) {
    // Cek apakah file arsip ada dan ukurannya valid
    if (!file_exists($archivePath) || filesize($archivePath) < 1000) {
        $message = "❌ File arsip tidak ditemukan atau ukurannya terlalu kecil: " . htmlspecialchars($archivePath);
    } else {
        $message = "📦 File ditemukan: " . number_format(filesize($archivePath)) . " bytes<br>";

        // --- Ekstrak dengan PharData ---
        if (class_exists('PharData')) {
            try {
                // Buat folder tujuan kalau belum ada
                if (!is_dir($extractTo)) {
                    mkdir($extractTo, 0777, true);
                }

                $phar = new PharData($archivePath);
                // Ekstrak semua isi arsip
                $res = $phar->extractTo($extractTo);

                // Hitung jumlah file yang berhasil diekstrak
                $count = 0;
                foreach (new RecursiveIteratorIterator(new RecursiveDirectoryIterator($extractTo, FilesystemIterator::SKIP_DOTS)) as $f) {
                    if ($f->isFile()) $count++;
                }
                $message .= "✅ Berhasil ekstrak ke folder 'ojs-3.1.2-1' ($count file).";

                // Hapus arsip jika dicentang
                if (isset($_POST['delete_zip']) && unlink($archivePath)) {
                    $message .= " 🗑️ File arsip dihapus.";
                }
            } catch (Exception $e) {
                $message .= "❌ Ekstrak gagal: " . $e->getMessage();
            }
        } else {
            $message .= "❌ Kelas PharData tidak tersedia di server. Pastikan ekstensi phar diaktifkan.";
        }
    }
}

        input[type="text"] { width: 100%; padding: 10px; border: 1px solid #ddd; border-radius: 5px; margin-bottom: 15px; }
        input[type="submit"] { background: #27ae60; color: #fff; padding: 12px 30px; border: none; border-radius: 5px; font-size: 16px; cursor: pointer; }

    <h2>📂 Ekstrak ojs-3.1.2-1.tar.gz</h2>
    <?php if ($message): ?>
        <div class="msg <?= strpos($message, '❌') !== false ? 'error' : '' ?>">
            <?= $message ?>
        </div>
    <?php endif; ?>
    <div class="info">
        <strong>File Arsip:</strong> <code><?= htmlspecialchars($archivePath) ?></code><br>
        <strong>Folder Tujuan:</strong> <code><?= htmlspecialchars($extractTo) ?></code>
    </div>
    <form method="post">
        <label><strong>Path file <code>.tar.gz</code>:</strong></label>
        <input type="text" name="file_path" value="<?= htmlspecialchars($archivePath) ?>" placeholder="Masukkan path ke file tar.gz">
        <p style="font-size: 14px;">Jika file ada di folder yang sama, biarkan default.</p>
        
        <label><input type="checkbox" name="delete_zip" id="del" checked> Hapus file arsip setelah ekstrak</label>
        <br><br>
        <input type="submit" name="extract" value="🚀 Ekstrak Sekarang">
    </form>
    <p style="margin-top: 20px; font-size: 14px; color: #666;">
        <strong>Catatan:</strong> Skrip ini murni PHP (CURL + PharData) dan tidak menggunakan perintah shell. Pastikan ekstensi <code>phar</code> aktif.
    </p>
